    Input / Output API¶

    The MassFlow.io module manages the file-system boundary. It handles loading spectra from mzML, MGF, MSP, and SQLite, and orchestrates the export of your final annotation results to CSV, JSON, Excel, and Parquet formats.

    MassFlow.io ¶

    I/O helpers for MassFlow spectral data.

    This module is the file-system boundary for MassFlow. It loads spectra from open interchange formats such as mzML, mzXML, MGF, and MSP, as well as SQLite databases. It exports search results to CSV and mzTab-M formats and provides utilities for writing provenance reports and open-format spectra.

    The loader is intentionally narrow in scope: it does not auto-convert proprietary vendor files and it does not sanitize metadata during import. Vendor raw formats are rejected with an actionable error, and metadata harmonization is deferred to :mod:MassFlow.processing after import.

    def is_store_input(path: Path) -> bool:
    """True when *path* is a MassFlow-native store (SQLite/Zarr).

    Zarr stores are directories carrying the ``.zgroup`` marker (or a ``.zarr``
    suffix). Existence of the underlying arrays is not verified here.

    def load_spectra(
    file_path: Path,
    file_format: Optional[str] = None,
    rejection_reporter: Optional[Callable[[str], None]] = None,
    *,
    assign_record_ids: bool = False,
    diagnostic_reporter: Optional[Callable[[RejectionEvent], None]] = None,
) -> Iterator[Spectrum]:
    """
    Load spectra from an open spectral file or a MassFlow-native store.

    The loader infers the format from ``file_path`` unless ``file_format`` is
    provided explicitly. Supported formats are mzML, mzXML, MGF, MSP, and SQLite
    (``db``/``sqlite``). Peak decoding is delegated to the backend loader.
    mzXML spectra also retain the original duration text by scan ID, before
    importer unit conversion. Metadata normalization and peak filtering happen
    later in :func:`MassFlow.processing.process_spectra`.

    Parameters
    ----------
    file_path : Path
        Path to the input file or directory. Bruker ``.d`` directories are
        treated as vendor raw inputs and rejected unless pre-converted.
    file_format : str, optional
        Explicit format override. Accepted values are ``mzml``, ``mzxml``,
        ``mgf``, ``msp``, ``db``, and ``sqlite``. If omitted, the format is
        inferred from the file extension.
    rejection_reporter : Callable[[str], None] or None
        Optional callback invoked with the rejection reason for every
        spectrum dropped by the validation layer (see
        :func:`_validate_spectra_iterator`).
    assign_record_ids : bool
        For raw library ingestion, clone each loader record with its source
        SHA-256 and zero-based loader ordinal. Preserve the first occurrence of
        an existing ID; missing or repeated IDs receive a content/ordinal ID.
        Original IDs remain in ``massflow_original_id``. Chemical identifiers
        are never used as record keys. Native stores are read without changes.
    diagnostic_reporter : Callable[[RejectionEvent], None] or None
        Optional typed observer for bounded failure-attribution diagnostics.

    Yields
    ------
    matchms.Spectrum
        Raw spectrum objects yielded one at a time.

    Raises
    ------
    UnsupportedVendorFormatError
        If ``file_path`` points to a vendor-specific raw format that must be
        converted to an open format before MassFlow ingestion.
    ValueError
        If the file format is unsupported.

    Notes
    -----
    ``matchms`` importers are called with ``metadata_harmonization=False`` so
    that MassFlow can apply its own processing pipeline in
    :mod:`MassFlow.processing`.
    """
    path = Path(file_path)
    if not path.exists():
        raise FileNotFoundError(f"Input path does not exist: {path}")

    # Step 1: Check for unsupported vendor formats
    if is_vendor_raw(path):
        raise UnsupportedVendorFormatError(VENDOR_FORMAT_ERROR_MESSAGE)

    # Step 2: Determine loading function
    ext = path.suffix.lower()
    fmt = (file_format or ext.lstrip(".")).lower()
    loader = None
    # Disable matchms internal harmonization to allow MassFlow's processing module to handle it

    if fmt == "mzml":
        loader = load_from_mzml(str(path), metadata_harmonization=False)
    elif fmt == "msp":
        loader = load_from_msp(str(path), metadata_harmonization=False)
    elif fmt == "mgf":
        loader = load_from_mgf(str(path), metadata_harmonization=False)
    elif fmt == "mzxml":
        loader = _mzxml_spectra_with_original_time(path)
    elif fmt in ["db", "sqlite"]:
        from MassFlow.storage import create_spectral_store

        # A hybrid database is a SQLite file whose peak arrays live in a
        # sibling `<stem>.zarr` store; attach it so metadata-only reads
        # still resolve fragment arrays.
        if path.with_suffix(".zarr").is_dir():
            store = create_spectral_store(path, backend="hybrid")
        else:
            store = create_spectral_store(path, backend="sqlite")
        loader = store.get_spectra()
    elif fmt == "zarr":
        from MassFlow.storage import create_spectral_store

        store = create_spectral_store(path, backend="zarr")
        loader = store.get_spectra()
    else:
        raise ValueError(f"Format '{fmt}' is not supported by MassFlow.")

    # Assign library identity before validation, so rejected records cannot
    # renumber later source records. Query loading keeps its existing contract.
    if assign_record_ids and fmt not in {"db", "sqlite", "zarr"}:
        loader = _assign_library_record_ids(loader, path)

    # Step 3: Yield spectra through the validation layer
    yield from _validate_spectra_iterator(
        loader, path, rejection_reporter, diagnostic_reporter
    )

    def promote_cache_resources(
    resources: Sequence[tuple[Optional[Path], Path]], parent: Path
) -> None:
    """Promote staged cache resources as one rollback-capable transaction.

    A ``None`` staged path means the destination is obsolete and should be
    removed as part of the same transaction. All staged resources and
    destinations must be on the same filesystem to support atomic replacement.

    Parameters
    ----------
    resources : Sequence[tuple[Optional[Path], Path]]
        Ordered staged-source and destination pairs. A missing source removes
        its destination transactionally.
    parent : Path
        Existing directory used to store temporary rollback backups.

    Returns
    -------
    None
        The operation completes without a return value. On failure, previous
        destinations are restored before the original exception is raised.

    Examples
    --------
    >>> promote_cache_resources([(Path("stage/store.db"), Path("cache/store.db"))], Path("cache"))
    """
    backup_directory = Path(
        tempfile.mkdtemp(prefix=".massflow-cache-backup-", dir=parent)
    )
    backups: list[tuple[Path, Path]] = []
    installed: list[Path] = []
    rollback_error: Optional[BaseException] = None
    try:
        for _, destination in resources:
            if destination.exists() or destination.is_symlink():
                backup_path = backup_directory / destination.name
                os.replace(destination, backup_path)
                backups.append((backup_path, destination))
        for staged_path, destination in resources:
            if staged_path is not None:
                os.replace(staged_path, destination)
                installed.append(destination)
    except BaseException:
        for destination in reversed(installed):
            try:
                _remove_path(destination)
            except BaseException as error:
                rollback_error = error
                break
        if rollback_error is None:
            for backup_path, destination in reversed(backups):
                if backup_path.exists() or backup_path.is_symlink():
                    try:
                        os.replace(backup_path, destination)
                    except BaseException as error:
                        rollback_error = error
                        break
        if rollback_error is not None:
            logger.exception(
                "Could not fully restore the prior prepared library cache; "
                "the backup remains at %s",
                backup_directory,
            )
            raise RuntimeError(
                "Prepared library promotion failed and rollback was incomplete; "
                f"the previous cache backup remains at {backup_directory}."
            ) from rollback_error
        raise
    finally:
        if rollback_error is None:
            shutil.rmtree(backup_directory, ignore_errors=True)

    def save_analysis_report(
    output_path: Path,
    report_data: dict[str, Any],
) -> None:
    """
    Save a YAML sidecar report describing the provenance of an analysis output.

    The report is intended to accompany a CSV results file and capture the
    configuration and runtime context that produced it. This helps keep result
    tables lightweight while preserving the scientific and procedural details
    needed to reproduce or interpret a run.

    Parameters
    ----------
    output_path : Path
        Destination path for the YAML report file.
    report_data : dict[str, Any]
        Serializable report content describing the analysis provenance.

    Returns
    -------
    None

    Raises
    ------
    OSError
        If the output directory cannot be created or the file cannot be
        written.
    """
    output_path.parent.mkdir(parents=True, exist_ok=True)

    report_payload = {
        "report_created_at": datetime.now(timezone.utc).isoformat(),
        **report_data,
    }

    with open(output_path, "w") as file_handle:
        yaml.safe_dump(
            report_payload,
            file_handle,
            sort_keys=False,
            allow_unicode=True,
        )

    logger.info(f"Analysis report saved to {output_path}")

    def save_match_results_to_mztab(
    results: list[dict[str, Any]],
    output_path: Path,
    query_spectra: Optional[Iterable[Spectrum]] = None,
    uncalibrated: bool = False,
) -> None:
    """
    Save annotation results in an experimental mzTab-shaped tab-separated format.

    This exporter is **not** mzTab-M standards-conformant. It writes MTD, SMH and
    SML sections with MassFlow-specific columns, but it does not write the
    mandatory mzTab-M controlled vocabulary, required table structure, or
    ``null`` missing-value encoding. CSV is the stable supported export. This
    function remains available for backward-compatible diagnostic interchange
    only.

    Parameters
    ----------
    results : list of dict
        Match result rows to export.
    output_path : Path
        The destination file path for the mzTab-M output.
    query_spectra : Optional[Iterable[Spectrum]]
        Full set of experimental query spectra.
    uncalibrated : bool
        True when the run's decoy null was empty and the q-values are the
        legacy ``1/N`` rank diagnostic (no FDR claim is supported); matched
        rows are then labeled ``Uncalibrated``.
    """
    df = _build_results_dataframe(results, query_spectra, uncalibrated=uncalibrated)
    if df is None:
        logger.warning("No results to save and no query_spectra provided.")
        return

    output_path.parent.mkdir(parents=True, exist_ok=True)

    with open(output_path, "w") as f:
        # 1. Metadata Section (MTD). Deliberately do not emit mzTab-version:
        # this file is mzTab-shaped, not mzTab-M standards-conformant.
        f.write(
            "MTD\tcomment\tMassFlow experimental tabular export; "
            "NOT mzTab-M standards-conformant\n"
        )
        f.write("MTD\tmassflow-export-id\tMassFlow_Export\n")
        f.write(
            f"MTD\ttitle\tMassFlow Annotation Results - {datetime.now().isoformat()}\n"
        )
        f.write("MTD\tdescription\tAutomated spectral annotation via MassFlow\n")

        # 2. Small Molecule Header (SMH)
        cols = df.columns
        header = "SMH\t" + "\t".join(cols) + "\n"
        f.write(header)

        # 3. Small Molecule List (SML)
        for row in df.iter_rows():
            row_vals = [str(v) if v is not None else "" for v in row]
            f.write("SML\t" + "\t".join(row_vals) + "\n")

    logger.warning(
        "Experimental mzTab-shaped export written to %s; this file is NOT "
        "mzTab-M standards-conformant. Use CSV for the supported export.",
        output_path,
    )

    @contextmanager
def temporary_directory(parent: Path, prefix: str) -> Iterator[Path]:
    """Create a temporary directory below a selected parent directory.

    Parameters
    ----------
    parent : Path
        Existing directory on the same filesystem as resources that may be
        atomically promoted from the temporary directory.
    prefix : str
        Prefix for the temporary directory name.

    Yields
    ------
    Path
        The temporary directory path. It is removed when the context exits.

    Examples
    --------
    >>> with temporary_directory(Path("results"), ".massflow-stage-") as stage:
    ...     (stage / "payload").write_text("ready")
    """
    with tempfile.TemporaryDirectory(prefix=prefix, dir=parent) as directory:
        yield Path(directory)
